# Path/file regex excluded from gitleaks scanning.
# These are merged into [allowlist] paths at runtime by scan.sh.
# Reasons: virtual fs, kernel/proc, container overlay/cache, package caches, vendored deps.

^/proc(/|$)
^/sys(/|$)
^/dev(/|$)
^/run(/|$)

/var/lib/containerd/
/var/lib/buildkit/
/var/lib/kubelet/pods/
/var/lib/snapd/
